# Advanced Patterns

## Cross-Runtime Test Fixtures

Share one fixture between Node.js unit tests and browser e2e tests:

```ts
// test/fixtures/user.ts
export const userFixture = {
  id: Symbol.for('user.id'),
  name: 'Test User',
  createdAt: new Date('2024-01-01T00:00:00.000Z'),
  roles: new Map([['admin', true]]),
  permissions: new Set(['read', 'write']),
  greet() { return `Hello, ${this.name}`; },
};

// test/unit/user.test.ts (Node.js + Vitest)
import { stringify } from 'jsoneo';
import { userFixture } from './fixtures/user';

test('serialize fixture', () => {
  const serialized = stringify(userFixture);
  expect(serialized).toContain('Test User');
});

// test/e2e/user.test.ts (Playwright + Browser)
import { parse } from 'jsoneo';
import { userFixture } from '../fixtures/user';

test('deserialize in browser', async ({ page }) => {
  const serialized = stringify(userFixture); // from Node
  await page.evaluate((data) => {
    const restored = parse(data);
    expect(restored.name).toBe('Test User');
    expect(restored.greet()).toBe('Hello, Test User');
  }, serialized);
});
```

## Custom Serialization with toJSON / fromJSON

```ts
class CustomData {
  constructor(public data: unknown) {}

  toJSON() {
    return { __type: 'CustomData', payload: this.data };
  }

  static fromJSON(json: any) {
    return new CustomData(json.payload);
  }
}

const original = new CustomData({ nested: { value: 42 } });
const serialized = stringify(original);
// Output includes __type and payload

const restored = parse(serialized);
// restored instanceof CustomData === true
// restored.data.nested.value === 42
```

## Preserving Class Instances

```ts
class Point {
  constructor(
    public x: number,
    public y: number,
  ) {}

  distance() {
    return Math.sqrt(this.x ** 2 + this.y ** 2);
  }
}

const point = new Point(3, 4);
const serialized = stringify(point, { preserveClassConstructor: true });
const restored = parse(serialized);

restored instanceof Point; // true
restored.distance(); // 5
```

## Handling External Dependencies in Functions

```ts
// Module-level constants
const API_BASE = 'https://api.example.com';

function fetchUser(id: string) {
  return fetch(`${API_BASE}/users/${id}`).then((r) => r.json());
}

// Attach to function for self-reference
fetchUser.API_BASE = API_BASE;

const serialized = stringify({ fetchUser });
const restored = parse(serialized);

restored.fetchUser.API_BASE; // 'https://api.example.com'
```

## Symbol Key Handling

```ts
const obj = {
  [Symbol.for('metadata')]: 'value1',
  [Symbol.iterator]: function* () {
    yield 1;
    yield 2;
  },
  local: Symbol('description'),
};

const serialized = stringify(obj);
const restored = parse(serialized);

// Well-known and global symbols preserved
restored[Symbol.for('metadata')]; // 'value1'
restored[Symbol.iterator]; // function* () { yield 1; yield 2; }
```

## Circular Reference Patterns

```ts
// Self-reference
const a = { name: 'a' };
a.self = a;

// Mutual reference
const parent = { name: 'parent' };
const child = { name: 'child', parent };
parent.child = child;

// Both serialize and restore correctly
const serialized = stringify({ a, parent });
const restored = parse(serialized);

restored.a.self === restored.a; // true
restored.parent.child.parent === restored.parent; // true
```
